from datetime import date

def easter_sunday(year: int) -> date:
    """
    Compute the date of Easter Sunday for a given Gregorian year using the
    Meeus/Jones/Butcher algorithm.
    Accepts:
      - year: integer year (1..9999, Gregorian rules typically apply from 1583)
    Returns:
      - date: the date of Easter Sunday in the given year
    """
    # Algorithm constants and computations
    a = year % 19
    b = year // 100
    c = year % 100
    d = b // 4
    e = b % 4
    f = (b + 8) // 25
    g = (b - f + 1) // 3

    i = c // 4
    k = c % 4
    l = (32 + 2 * e + 2 * i - h - k) % 7

    month = (h + l - 7 * m + 114) // 31  # 3 = March, 4 = April

    return date(year, month, day)
