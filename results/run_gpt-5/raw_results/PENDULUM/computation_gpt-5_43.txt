import pendulum

def is_weekend(d: pendulum.Date) -> bool:
    """
    Determine if the given pendulum.Date falls on a weekend.

    Args:
        d (pendulum.Date): The date to check.

    Returns:
        bool: True if the date is Saturday or Sunday, otherwise False.
    """
    return d.is_weekend()
