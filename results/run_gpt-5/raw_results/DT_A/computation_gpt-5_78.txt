Let's think step by step to solve this problem.
1. Parse the input date and compute the day-of-year (N) for the given date.
2. Use the NOAA sunrise algorithm to approximate the sun’s position for sunrise:
   - Compute an approximate time (t) based on longitude and day-of-year.
   - Calculate the Sun’s mean anomaly (M).
   - Compute the Sun’s true longitude (L) and right ascension (RA).
   - Compute the Sun’s declination (sinDec, cosDec).
   - Compute the hour angle (H) for sunrise using the standard zenith 90.833 degrees.
3. Convert the hour angle into Universal Time (UT), normalize it to 0–24 hours.
4. Build a UTC datetime for the given date and UT, then convert to the provided time zone.
5. Return the localized sunrise datetime. Raise an error if the sun never rises or never sets on that date at that latitude.

```python
from datetime import date, datetime, timezone, timedelta
from zoneinfo import ZoneInfo
import math

def _normalize_angle_deg(x: float) -> float:
    # Normalize angle to [0, 360)
    return x % 360.0

def _deg_to_rad(d: float) -> float:
    return math.radians(d)

def _rad_to_deg(r: float) -> float:
    return math.degrees(r)

def calculate_sunrise(d: date, latitude: float, longitude: float, tz: ZoneInfo) -> datetime:
    """
    Calculate the local sunrise time for a given date and geographic coordinates using the NOAA algorithm.
    Inputs:
      - d: date (calendar date for which to compute sunrise)
      - latitude: float latitude in degrees (positive north, negative south)
      - longitude: float longitude in degrees (positive east, negative west)
      - tz: zoneinfo.ZoneInfo time zone for the output
    Returns:
      - datetime localized to tz representing the sunrise time on date d
    Raises:
      - ValueError if the sun never rises or never sets on that date at the given latitude.
    """
    # Constants
    zenith = 90.833  # Official sunrise zenith in degrees (includes refraction)
    day_of_year = (d - date(d.year, 1, 1)).days + 1

    lng_hour = longitude / 15.0

    # Approximate time for sunrise: 6 AM local solar time equivalent
    t = day_of_year + ((6.0 - lng_hour) / 24.0)

    # Sun's mean anomaly
    M = (0.9856 * t) - 3.289

    # Sun's true longitude
    L = M + (1.916 * math.sin(_deg_to_rad(M))) + (0.020 * math.sin(_deg_to_rad(2 * M))) + 282.634
    L = _normalize_angle_deg(L)

    # Sun's right ascension
    RA = _rad_to_deg(math.atan(0.91764 * math.tan(_deg_to_rad(L))))
    RA = _normalize_angle_deg(RA)

    # Adjust RA to be in the same quadrant as L
    L_quadrant = math.floor(L / 90.0) * 90.0
    RA_quadrant = math.floor(RA / 90.0) * 90.0
    RA = RA + (L_quadrant - RA_quadrant)

    # Convert RA to hours
    RA_hours = RA / 15.0

    # Sun's declination
    sinDec = 0.39782 * math.sin(_deg_to_rad(L))
    cosDec = math.cos(math.asin(sinDec))

    # Hour angle
    cosH = (math.cos(_deg_to_rad(zenith)) - (sinDec * math.sin(_deg_to_rad(latitude)))) / (cosDec * math.cos(_deg_to_rad(latitude)))

    # Check for polar day/night conditions
    if cosH > 1.0:
        raise ValueError("Sun never rises on this date at the given latitude.")
    if cosH < -1.0:
        raise ValueError("Sun never sets on this date at the given latitude.")

    # For sunrise, hour angle H is 360 - arccos(cosH)
    H = 360.0 - _rad_to_deg(math.acos(cosH))
    H_hours = H / 15.0

    # Local mean time of rising
    T = H_hours + RA_hours - (0.06571 * t) - 6.622

    # Convert to UT
    UT = T - lng_hour
    # Normalize UT to [0, 24)
    UT = (UT % 24.0 + 24.0) % 24.0

    # Build UTC datetime for the given date at UT
    hours = int(UT)
    minutes_float = (UT - hours) * 60.0
    minutes = int(minutes_float)
    seconds = int(round((minutes_float - minutes) * 60.0))

    # Handle rounding that might push seconds to 60
    if seconds >= 60:
        seconds -= 60
        minutes += 1
    if minutes >= 60:
        minutes -= 60
        hours += 1
    if hours >= 24:
        # Wrap to next day in UTC
        dt_utc_date = d + timedelta(days=1)
        hours -= 24
    else:
        dt_utc_date = d

    dt_utc = datetime(dt_utc_date.year, dt_utc_date.month, dt_utc_date.day, hours, minutes, seconds, tzinfo=timezone.utc)

    # Convert to target timezone
    dt_local = dt_utc.astimezone(tz)
    return dt_local
```
