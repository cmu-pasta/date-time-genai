    Calculates the age in years, months, and days between two datetime objects.

    Args:
        start_date (datetime): The earlier date (e.g., birth date).
        end_date (datetime): The later date (e.g., current date).

    years = end_date.year - start_date.year
    months = end_date.month - start_date.month

        days_in_prev_month = temp_date_for_prev_month.day
        days += days_in_prev_month
        months -= 1 # Adjust month since we borrowed days

        years -= 1 # Adjust year since we borrowed months
