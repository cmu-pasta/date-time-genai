def gregorian_to_hebrew(dt: datetime) -> int:

    # This represents 1 Tishrei 1 in Hebrew calendar

    year = dt.year
    month = dt.month
    day = dt.day

    if month <= 2:
        year -= 1
        month += 12

    b = 2 - a + a // 4
