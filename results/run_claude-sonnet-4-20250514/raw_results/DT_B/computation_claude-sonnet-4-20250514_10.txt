from datetime import date

def date_to_julian_day(input_date: date) -> int:
    """
    Convert a date to Julian day number.
    Julian day number is the continuous count of days since 
    January 1, 4713 BCE in the proleptic Julian calendar.
    """
    year = input_date.year
    month = input_date.month
    day = input_date.day
    
    # Adjust for the algorithm: treat Jan and Feb as months 13 and 14 of previous year
    if month <= 2:
        year -= 1
        month += 12
    
    # Calculate Julian day number using the standard algorithm
    a = year // 100
    b = 2 - a + (a // 4)
    
    # Calculate Julian day number
    jdn = int(365.25 * (year + 4716)) + int(30.6001 * (month + 1)) + day + b - 1524
    
    return jdn
