from datetime import date, datetime, time, timedelta, timezone
from zoneinfo import ZoneInfo
import math

def get_sunset_time(date_input: date, latitude: float, longitude: float, tzinfo: ZoneInfo) -> datetime:
    """Compute the sunset time (approximate, civil) for a given date and location."""
    # Step 1: Calculate the day of the year (N)
    N = date_input.timetuple().tm_yday

    # Step 2: Convert latitude to radians
    lat_rad = math.radians(latitude)

    # Step 3: Approximate the sun's mean anomaly and other solar parameters (for sunset)
    lng_hour = longitude / 15

    # Sun's mean anomaly
    t = N + ((18 - lng_hour) / 24)  # 18h is for sunset
    M = (0.9856 * t) - 3.289

    # Sun's true longitude
    L = M + (1.916 * math.sin(math.radians(M))) + (0.020 * math.sin(math.radians(2 * M))) + 282.634
    L = L % 360

    # Sun's right ascension
    RA = math.degrees(math.atan(0.91764 * math.tan(math.radians(L))))
    RA = RA % 360

    # Right ascension value needs to be in the same quadrant as L
    Lquadrant  = (math.floor( L/90)) * 90
    RAquadrant = (math.floor(RA/90)) * 90
    RA = RA + (Lquadrant - RAquadrant)
    RA = RA / 15  # Convert to hours

    # Sun's declination
    sinDec = 0.39782 * math.sin(math.radians(L))
    cosDec = math.cos(math.asin(sinDec))

    # Sun's local hour angle
    cosH = (math.cos(math.radians(90.833)) - (sinDec * math.sin(lat_rad))) / (cosDec * math.cos(lat_rad))

    # If the sun never sets or never rises on this location on the given date
    if cosH < -1:
        # Sun never sets (polar day)
        raise ValueError("The sun never sets on this date at this location.")
    elif cosH > 1:
        # Sun never rises (polar night)
        raise ValueError("The sun never rises on this date at this location.")

    # Sunset hour angle
    H = math.degrees(math.acos(cosH)) / 15  # sunset = use +H

    # Local mean time of setting
    T = H + RA - (0.06571 * t) - 6.622

    # Adjust back to UTC
    UT = (T - lng_hour) % 24

    # Step 4: Build UTC datetime for sunset
    hour = int(UT)
    minute = int((UT - hour) * 60)
    second = int((((UT - hour) * 60) - minute) * 60)
    sunset_utc = datetime.combine(date_input, time(hour=hour, minute=minute, second=second), tzinfo=timezone.utc)

    # Step 5: Convert UTC sunset to target time zone
    sunset_local = sunset_utc.astimezone(tzinfo)

    # Step 6: Return sunset time as timezone-aware datetime
    return sunset_local
